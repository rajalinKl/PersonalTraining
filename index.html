Google Sheets Backend Setup Guide

Connect this web app to your Google Sheet in under 5 minutes so you can manage clients and logs right from your spreadsheet.

1 Create Your Google Sheet

Create a new Google Sheet named ProTrainerDB. Create two tabs (worksheets) inside it:

  • Tab 1 named Clients (Columns: id, name, weight, bodyFat, calories, protein, dietNotes)
  • Tab 2 named Workouts (Columns: id, clientId, clientName, title, date, exercisesJson, notes)
2 Add the Apps Script Code (Fixed for CORS & Parameters)

In your Google Sheet, click on Extensions > Apps Script. Clear any existing code and paste the following updated script:

function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var action = e && e.parameter ? e.parameter.action : 'getData';
    
    if (action === 'getData') {
      var clientSheet = ss.getSheetByName('Clients');
      var workoutSheet = ss.getSheetByName('Workouts');
      
      var clients = [];
      if (clientSheet && clientSheet.getLastRow() > 1) {
        var cData = clientSheet.getDataRange().getValues();
        clients = parseSheetData(cData);
      }
      
      var workouts = [];
      if (workoutSheet && workoutSheet.getLastRow() > 1) {
        var wData = workoutSheet.getDataRange().getValues();
        workouts = parseSheetData(wData);
      }
      
      return ContentService.createTextOutput(JSON.stringify({status: 'success', clients: clients, workouts: workouts}))
        .setMimeType(ContentService.MimeType.JSON);
    }
    return ContentService.createTextOutput(JSON.stringify({status: 'error', message: 'Invalid action'})).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status: 'error', message: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var postData = JSON.parse(e.postData.contents);
    var action = postData.action;
    
    if (action === 'saveData') {
      var cSheet = ss.getSheetByName('Clients');
      if (!cSheet) { cSheet = ss.insertSheet('Clients'); }
      cSheet.clear();
      cSheet.appendRow(['id', 'name', 'weight', 'bodyFat', 'calories', 'protein', 'dietNotes']);
      if (postData.clients) {
        postData.clients.forEach(function(c) {
          cSheet.appendRow([c.id || '', c.name || '', c.weight || 0, c.bodyFat || 0, c.calories || 0, c.protein || 0, c.dietNotes || '']);
        });
      }
      
      var wSheet = ss.getSheetByName('Workouts');
      if (!wSheet) { wSheet = ss.insertSheet('Workouts'); }
      wSheet.clear();
      wSheet.appendRow(['id', 'clientId', 'clientName', 'title', 'date', 'exercisesJson', 'notes']);
      if (postData.workouts) {
        postData.workouts.forEach(function(w) {
          wSheet.appendRow([w.id || '', w.clientId || '', w.clientName || '', w.title || '', w.date || '', JSON.stringify(w.exercises || []), w.notes || '']);
        });
      }
      
      return ContentService.createTextOutput(JSON.stringify({status: 'success'}))
        .setMimeType(ContentService.MimeType.JSON);
    }
    return ContentService.createTextOutput(JSON.stringify({status: 'error', message: 'Invalid POST action'})).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status: 'error', message: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}

function parseSheetData(rows) {
  var headers = rows[0];
  var result = [];
  for (var i = 1; i < rows.length; i++) {
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      obj[headers[j]] = rows[i][j];
    }
    result.push(obj);
  }
  return result;
}
